   127→
   128→export function MainDashboard() {
   129→  const [prompt, setPrompt] = useState('')
   130→  const [files, setFiles] = useState<File[]>([])
   131→  const [selectedFormat, setSelectedFormat] = useState('auto')
   132→  const [isGenerating, setIsGenerating] = useState(false)
   133→  const [showResultDialog, setShowResultDialog] = useState(false)
   134→  const [generationStages, setGenerationStages] = useState<GenerationStage[]>([])
   135→  const [currentArtifact, setCurrentArtifact] = useState<ArtifactPreview | null>(null)
   136→  const [dragActive, setDragActive] = useState(false)
   137→
   138→  // Handle file upload
   139→  const handleFiles = useCallback((newFiles: FileList | null) => {
   140→    if (!newFiles) return
   141→    
   142→    const fileArray = Array.from(newFiles)
   143→    setFiles(prev => [...prev, ...fileArray])
   144→  }, [])
   145→
   146→  const handleDrag = useCallback((e: React.DragEvent) => {
   147→    e.preventDefault()
   148→    e.stopPropagation()
   149→    if (e.type === 'dragenter' || e.type === 'dragover') {
   150→      setDragActive(true)
   151→    } else if (e.type === 'dragleave') {
   152→      setDragActive(false)
   153→    }
   154→  }, [])
   155→
   156→  const handleDrop = useCallback((e: React.DragEvent) => {
   157→    e.preventDefault()
   158→    e.stopPropagation()
   159→    setDragActive(false)
   160→    handleFiles(e.dataTransfer.files)
   161→  }, [handleFiles])
   162→
   163→  const removeFile = (index: number) => {
   164→    setFiles(prev => prev.filter((_, i) => i !== index))
   165→  }
   166→
   167→  // Handle generation
   168→  const handleGenerate = async () => {
   169→    if (!prompt.trim()) return
   170→
   171→    setIsGenerating(true)
   172→    
   173→    // Initialize generation stages
   174→    const stages: GenerationStage[] = [
   175→      { id: '1', label: 'Understanding request', status: 'pending' },
   176→      { id: '2', label: 'Reading files', status: 'pending' },
   177→      { id: '3', label: 'Planning artifact', status: 'pending' },
   178→      { id: '4', label: 'Generating content', status: 'pending' },
   179→      { id: '5', label: 'Creating visuals', status: 'pending' },
   180→      { id: '6', label: 'Formatting document', status: 'pending' },
   181→      { id: '7', label: 'Checking quality', status: 'pending' },
   182→      { id: '8', label: 'Finalizing', status: 'pending' },
   183→    ]
   184→    setGenerationStages(stages)
   185→
   186→    try {
   187→      // Simulate generation stages (in production, these would be real API calls)
   188→      for (let i = 0; i < stages.length; i++) {
   189→        await new Promise(resolve => setTimeout(resolve, 800 + Math.random() * 1200))
   190→        
   191→        setGenerationStages(prev => 
   192→          prev.map(stage => 
   193→            stage.id === String(i + 1) 
   194→              ? { ...stage, status: 'completed' }
   195→              : stage
   196→          )
   197→        )
   198→        
   199→        if (i < stages.length - 1) {
   200→          setGenerationStages(prev =>
   201→            prev.map(stage =>
   202→              stage.id === String(i + 2)
   203→                ? { ...stage, status: 'active' }
   204→                : stage
   205→            )
   206→          )
   207→        }
   208→      }
   209→
   210→      // Create artifact preview
   211→      const artifact: ArtifactPreview = {
   212→        id: crypto.randomUUID(),
   213→        title: extractTitle(prompt),
   214→        type: detectArtifactType(prompt),
   215→        format: selectedFormat === 'auto' ? 'DOCX' : selectedFormat,
   216→        status: 'completed',
   217→        createdAt: new Date(),
   218→      }
   219→
   220→      setCurrentArtifact(artifact)
   221→      setShowResultDialog(true)
   222→    } catch (error) {
   223→      console.error('Generation failed:', error)
   224→      setGenerationStages(prev =>
   225→        prev.map(stage =>
   226→          stage.status === 'active'
   227→            ? { ...stage, status: 'error', detail: 'Generation failed' }
   228→            : stage
   229→        )
   230→      )
   231→    } finally {
   232→      setIsGenerating(false)
   233→    }
   234→  }
   235→
   236→  // Helper functions
   237→  const extractTitle = (text: string): string => {
   238→    const words = text.split(' ').slice(0, 8).join(' ')
   239→    return words.length > 50 ? words + '...' : words
   240→  }
   241→
   242→  const detectArtifactType = (text: string): string => {
   243→    const lower = text.toLowerCase()
   244→    if (lower.includes('proposal') || lower.includes('business')) return 'Proposal'
   245→    if (lower.includes('lesson') || lower.includes('course')) return 'Lesson Plan'
   246→    if (lower.includes('invoice')) return 'Invoice'
   247→    if (lower.includes('resume') || lower.includes('cv')) return 'Resume'
   248→    if (lower.includes('presentation') || lower.includes('slide')) return 'Presentation'
   249→    if (lower.includes('spreadsheet') || lower.includes('excel') || lower.includes('budget')) return 'Spreadsheet'
   250→    if (lower.includes('report')) return 'Report'
   251→    return 'Document'
   252→  }
   253→
   254→  const handleExampleClick = (examplePrompt: string) => {
   255→    setPrompt(examplePrompt)
   256→  }
   257→
   258→  return (
   259→    <div className="min-h-screen bg-background">
   260→      {/* Hero Section */}
   261→      <section className="relative overflow-hidden border-b bg-gradient-to-br from-background via-background to-muted/30">
   262→        <div className="absolute inset-0 bg-[linear-gradient(to_right,#8882_1px,transparent_1px),linear-gradient(to_bottom,#8882_1px,transparent_1px)] bg-[size:24px_24px] [mask-image:radial-gradient(ellipse_60%_50%_at_50%_0%,#000_70%,transparent_100%)]" />
   263→        
   264→        <div className="container relative mx-auto px-4 py-16 lg:py-24">
   265→          <div className="mx-auto max-w-4xl text-center">
   266→            {/* Badge */}
   267→            <Badge variant="secondary" className="mb-6 px-4 py-1.5 text-sm">
   268→              <Sparkles className="mr-2 h-3.5 w-3.5" />
   269→              AI-Powered Artifact Generation
   270→            </Badge>
   271→
   272→            {/* Headline */}
   273→            <h1 className="mb-6 text-4xl font-bold tracking-tight sm:text-5xl lg:text-6xl">
   274→              Describe what you need.
   275→              <br />
   276→              <span className="bg-gradient-to-r from-primary to-primary/70 bg-clip-text text-transparent">
   277→                Filo creates it.
   278→              </span>
   279→            </h1>
   280→
   281→            {/* Subheadline */}
   282→            <p className="mx-auto mb-10 max-w-2xl text-lg text-muted-foreground">
   283→              Transform your ideas into professional documents, spreadsheets, presentations, and more. 
   284→              No templates. No formatting. Just describe what you want.
   285→            </p>
   286→
   287→            {/* Main Input */}
   288→            <div className="mx-auto max-w-3xl">
   289→              <div className="rounded-xl border bg-card p-2 shadow-lg">
   290→                <Textarea
   291→                  placeholder="What do you want to create? Be specific about what you need..."
   292→                  value={prompt}
   293→                  onChange={(e) => setPrompt(e.target.value)}
   294→                  className="min-h-[120px] resize-none border-0 bg-transparent text-base focus-visible:ring-0"
   295→                  disabled={isGenerating}
   296→                />
   297→
   298→                {/* Files area */}
   299→                {(files.length > 0 || dragActive) && (
   300→                  <div
   301→                    className={`mx-2 mb-2 rounded-lg border-2 border-dashed p-4 transition-colors ${
   302→                      dragActive ? 'border-primary bg-primary/5' : 'border-muted-foreground/25'
   303→                    }`}
   304→                    onDragEnter={handleDrag}
   305→                    onDragLeave={handleDrag}
   306→                    onDragOver={handleDrag}
   307→                    onDrop={handleDrop}
   308→                  >
   309→                    {files.length > 0 ? (
   310→                      <div className="space-y-2">
   311→                        {files.map((file, index) => (
   312→                          <div key={index} className="flex items-center gap-2 rounded bg-muted/50 p-2">
   313→                            <FileText className="h-4 w-4 text-muted-foreground" />
   314→                            <span className="flex-1 truncate text-sm">{file.name}</span>
   315→                            <span className="text-xs text-muted-foreground">
   316→                              {(file.size / 1024).toFixed(1)} KB
   317→                            </span>
   318→                            <Button
   319→                              variant="ghost"
   320→                              size="icon"
   321→                              className="h-6 w-6"
   322→                              onClick={() => removeFile(index)}
   323→                              disabled={isGenerating}
   324→                            >
   325→                              <X className="h-3 w-3" />
   326→                            </Button>
   327→                          </div>
   328→                        ))}
   329→                        <label>
   330→                          <span className="cursor-pointer text-sm text-muted-foreground hover:text-foreground">
   331→                            + Add more files
   332→                          </span>
   333→                          <input
   334→                            type="file"
   335→                            multiple
   336→                            className="hidden"
   337→                            onChange={(e) => handleFiles(e.target.files)}
   338→                            disabled={isGenerating}
   339→                          />
   340→                        </label>
   341→                      </div>
   342→                    ) : (
   343→                      <div className="flex flex-col items-center justify-center py-4 text-sm text-muted-foreground">
   344→                        <Upload className="mb-2 h-8 w-8" />
   345→                        <p>Drop files here or click to upload</p>
   346→                        <input
   347→                          type="file"
   348→                          multiple
   349→                          className="hidden"
   350→                          onChange={(e) => handleFiles(e.target.files)}
   351→                          disabled={isGenerating}
   352→                        />
   353→                        <label className="mt-2 cursor-pointer text-xs text-primary hover:underline">
   354→                          Browse files
   355→                        </label>
   356→                      </div>
   357→                    )}
   358→                  </div>
   359→                )}
   360→
   361→                {/* Actions */}
   362→                <div className="flex flex-wrap items-center gap-3 border-t pt-3">
   363→                  {!files.length && (
   364→                    <label className="cursor-pointer">
   365→                      <Button variant="outline" size="sm" className="gap-2" asChild>
   366→                        <>
   367→                          <Upload className="h-4 w-4" />
   368→                          Add files
   369→                        </>
   370→                      </Button>
   371→                      <input
   372→                        type="file"
   373→                        multiple
   374→                        className="hidden"
   375→                        onChange={(e) => handleFiles(e.target.files)}
   376→                        disabled={isGenerating}
   377→                      />
   378→                    </label>
   379→                  )}
   380→
   381→                  <Select value={selectedFormat} onValueChange={setSelectedFormat}>
   382→                    <SelectTrigger className="w-[180px]" disabled={isGenerating}>
   383→                      <SelectValue placeholder="Output format" />
   384→                    </SelectTrigger>
   385→                    <SelectContent>
   386→                      {outputFormats.map(format => (
   387→                        <SelectItem key={format.value} value={format.value}>
   388→                          <div className="flex flex-col items-start">
   389→                            <span>{format.label}</span>
   390→                            <span className="text-xs text-muted-foreground">{format.description}</span>
   391→                          </div>
   392→                        </SelectItem>
   393→                      ))}
   394→                    </SelectContent>
   395→                  </Select>
   396→
   397→                  <div className="ml-auto flex gap-2">
   398→                    <Button variant="ghost" onClick={() => { setPrompt(''); setFiles([]); }} disabled={isGenerating}>
   399→                      Clear
   400→                    </Button>
   401→                    <Button
   402→                      onClick={handleGenerate}
   403→                      disabled={!prompt.trim() || isGenerating}
   404→                      className="gap-2 min-w-[120px]"
   405→                    >
   406→                      {isGenerating ? (
   407→                        <>
   408→                          <Loader2 className="h-4 w-4 animate-spin" />
   409→                          Creating...
   410→                        </>
   411→                      ) : (
   412→                        <>
   413→                          <Sparkles className="h-4 w-4" />
   414→                          Create
   415→                        </>
   416→                      )}
   417→                    </Button>
   418→                  </div>
   419→                </div>
   420→              </div>
   421→            </div>
   422→          </div>
   423→        </div>
   424→      </section>
   425→
   426→      {/* Progress Section */}
   427→      {isGenerating && (
   428→        <section className="border-b bg-muted/30">
   429→          <div className="container mx-auto px-4 py-8">
   430→            <div className="mx-auto max-w-2xl">
   431→              <div className="mb-4 flex items-center justify-between">
   432→                <h3 className="font-semibold">Generating your artifact</h3>
   433→                <Badge variant="secondary">
   434→                  {generationStages.filter(s => s.status === 'completed').length}/{generationStages.length}
   435→                </Badge>
   436→              </div>
   437→              
   438→              <Progress 
   439→                value={(generationStages.filter(s => s.status === 'completed').length / generationStages.length) * 100} 
   440→                className="mb-6"
   441→              />
   442→
   443→              <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
   444→                {generationStages.map(stage => (
   445→                  <div
   446→                    key={stage.id}
   447→                    className={`flex items-center gap-2 rounded-lg border p-3 transition-colors ${
   448→                      stage.status === 'active' ? 'border-primary bg-primary/5' :
   449→                      stage.status === 'completed' ? 'border-green-200 bg-green-50 dark:border-green-900 dark:bg-green-950' :
   450→                      stage.status === 'error' ? 'border-red-200 bg-red-50 dark:border-red-900 dark:bg-red-950' :
   451→                      'border-transparent'
   452→                    }`}
   453→                  >
   454→                    {stage.status === 'active' && <Loader2 className="h-4 w-4 animate-spin text-primary" />}
   455→                    {stage.status === 'completed' && <Check className="h-4 w-4 text-green-600" />}
   456→                    {stage.status === 'error' && <AlertCircle className="h-4 w-4 text-red-600" />}
   457→                    {stage.status === 'pending' && <div className="h-4 w-4 rounded-full border-2 border-muted-foreground/20" />}
   458→                    
   459→                    <span className={`text-sm ${
   460→                      stage.status === 'active' ? 'font-medium text-primary' :
   461→                      stage.status === 'completed' ? 'text-green-700 dark:text-green-400' :
   462→                      stage.status === 'error' ? 'text-red-700 dark:text-red-400' :
   463→                      'text-muted-foreground'
   464→                    }`}>
   465→                      {stage.label}
   466→                    </span>
   467→                  </div>
   468→                ))}
   469→              </div>
   470→            </div>
   471→          </div>
   472→        </section>
   473→      )}
   474→
   475→      {/* Example Prompts */}
   476→      <section className="py-12 lg:py-16">
   477→        <div className="container mx-auto px-4">
   478→          <div className="mb-8 text-center">
   479→            <h2 className="text-2xl font-bold tracking-tight">What can you create?</h2>
   480→            <p className="mt-2 text-muted-foreground">
   481→              Click any example to get started, or describe your own needs above
   482→            </p>
   483→          </div>
   484→
   485→          <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   486→            {examplePrompts.map((example, index) => (
   487→              <Card
   488→                key={index}
   489→                className="group cursor-pointer transition-all hover:border-primary/50 hover:shadow-md"
   490→                onClick={() => handleExampleClick(example.prompt)}
   491→              >
   492→                <CardHeader className="pb-3">
   493→                  <div className="flex items-center gap-3">
   494→                    <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-primary/10 transition-colors group-hover:bg-primary/20">
   495→                      <example.icon className="h-5 w-5 text-primary" />
   496→                    </div>
   497→                    <div>
   498→                      <CardTitle className="text-base">{example.title}</CardTitle>
   499→                      <Badge variant="secondary" className="mt-1 text-xs capitalize">
   500→                        {example.category}
   501→                      </Badge>
   502→                    </div>
   503→                  </div>
   504→                </CardHeader>
   505→                <CardContent>
   506→                  <CardDescription className="line-clamp-2 text-sm">
   507→                    {example.prompt}
   508→                  </CardDescription>
   509→                  <div className="mt-3 flex items-center text-sm font-medium text-primary opacity-0 transition-opacity group-hover:opacity-100">
   510→                    Use this example
   511→                    <ArrowRight className="ml-1 h-4 w-4" />
   512→                  </div>
   513→                </CardContent>
   514→              </Card>
   515→            ))}
   516→          </div>
   517→        </div>
   518→      </section>
   519→
   520→      {/* Capabilities */}
   521→      <section className="border-t bg-muted/30 py-12 lg:py-16">
   522→        <div className="container mx-auto px-4">
   523→          <div className="mb-10 text-center">
   524→            <h2 className="text-2xl font-bold tracking-tight">Professional outputs</h2>
   525→            <p className="mt-2 text-muted-foreground">
   526→              Every artifact is formatted, designed, and ready to use
   527→            </p>
   528→          </div>
   529→
   530→          <div className="grid gap-6 md:grid-cols-2 lg:grid-cols-4">
   531→            {[
   532→              {
   533→                icon: FileText,
   534→                title: 'Documents',
   535→                formats: ['DOCX', 'PDF'],
   536→                description: 'Reports, proposals, plans, research papers with proper formatting',
   537→              },
   538→              {
   539→                icon: Table,
   540→                title: 'Spreadsheets',
   541→                formats: ['XLSX', 'CSV'],
   542→                description: 'Budgets, trackers, schedules with formulas and calculations',
   543→              },
   544→              {
   545→                icon: Presentation,
   546→                title: 'Presentations',
   547→                formats: ['PPTX'],
   548→                description: 'Engaging slides with consistent design and visuals',
   549→              },
   550→              {
   551→                icon: BarChart3,
   552→                title: 'Charts & Diagrams',
   553→                formats: ['Embedded'],
   554→                description: 'Data visualizations, flowcharts, org charts, and more',
   555→              },
   556→            ].map((capability, index) => (
   557→              <Card key={index} className="text-center">
   558→                <CardContent className="pt-6">
   559→                  <div className="mx-auto mb-4 flex h-14 w-14 items-center justify-center rounded-xl bg-primary/10">

   561→                  </div>
   562→                  <h3 className="mb-2 font-semibold">{capability.title}</h3>
   563→                  <div className="mb-2 flex justify-center gap-1">
   564→                    {capability.formats.map(format => (

   569→                  </div>
   570→                  <p className="text-sm text-muted-foreground">{capability.description}</p>
   571→                </CardContent>

   574→          </div>
   575→        </div>
   576→      </section>
