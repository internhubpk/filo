   172→  useEffect(() => {
   173→    const savedPrompt = localStorage.getItem('filo_draft_prompt')
   174→    if (savedPrompt) {
   175→      setPrompt(savedPrompt)
   176→    }
   177→    
   178→    // Check for existing session
   179→    const savedUser = localStorage.getItem('filo_user')
   180→    if (savedUser) {
   181→      try {
   182→        setUser(JSON.parse(savedUser))
   183→      } catch {
   184→        localStorage.removeItem('filo_user')
   185→      }
   186→    }
   187→  }, [])
   188→

   191→    if (prompt.trim()) {
   192→      localStorage.setItem('filo_draft_prompt', prompt)

   194→      localStorage.removeItem('filo_draft_prompt')
   195→    }

   200→    if (!newFiles) return

   203→    setFiles(prev => [...prev, ...fileArray])

   207→    e.preventDefault()
   208→    e.stopPropagation()

   210→      setDragActive(true)
   211→    } else if (e.type === 'dragleave') {
   212→      setDragActive(false)

   214→  }, [])
   215→
   216→  const handleDrop = useCallback((e: React.DragEvent) => {
   217→    e.preventDefault()
   218→    e.stopPropagation()
   219→    setDragActive(false)
   220→    handleFiles(e.dataTransfer.files)
   221→  }, [handleFiles])
   222→
   223→  const removeFile = (index: number) => {
   224→    setFiles(prev => prev.filter((_, i) => i !== index))
   225→  }
   226→
   227→  // Auth Handlers
   228→  const handleLogin = async (e: React.FormEvent) => {
   229→    e.preventDefault()
   230→    setError(null)
   231→    
   232→    if (!loginEmail.trim() || !loginPassword.trim()) {
   233→      setError('Please fill in all fields')
   234→      return
   235→    }
   236→
   237→    setIsLoggingIn(true)
   238→    
   239→    try {
   240→      // Simulate API call - replace with real auth
   241→      await new Promise(resolve => setTimeout(resolve, 1000))
   242→      
   243→      const newUser: User = {
   244→        id: crypto.randomUUID(),
   245→        email: loginEmail,
   246→        name: loginEmail.split('@')[0],
   247→      }
   248→      
   249→      setUser(newUser)
   250→      localStorage.setItem('filo_user', JSON.stringify(newUser))
   251→      setShowLoginModal(false)
   252→      setLoginEmail('')
   253→      setLoginPassword('')
   254→    } catch (err) {
   255→      setError('Invalid credentials. Please try again.')
   256→    } finally {
   257→      setIsLoggingIn(false)
   258→    }
   259→  }
   260→
   261→  const handleSignup = async (e: React.FormEvent) => {
   262→    e.preventDefault()
   263→    setError(null)
   264→    
   265→    if (!signupName.trim() || !signupEmail.trim() || !signupPassword.trim()) {
   266→      setError('Please fill in all fields')
   267→      return
   268→    }
   269→
   270→    if (signupPassword.length < 6) {
   271→      setError('Password must be at least 6 characters')
   272→      return
   273→    }
   274→
   275→    setIsSigningUp(true)
   276→    
   277→    try {
   278→      // Simulate API call - replace with real auth
   279→      await new Promise(resolve => setTimeout(resolve, 1200))
   280→      
   281→      const newUser: User = {
   282→        id: crypto.randomUUID(),
   283→        email: signupEmail,
   284→        name: signupName,
   285→      }
   286→      
   287→      setUser(newUser)

   305→  // Handle generation
   306→  const handleGenerate = async () => {
   307→    if (!prompt.trim()) return
   308→    
   309→    // Require login for generation
   310→    if (!user) {
   311→      setShowLoginModal(true)
   312→      return
   313→    }
   314→
   315→    setIsGenerating(true)

   317→    
   318→    // Initialize generation stages
   319→    const stages: GenerationStage[] = [
   320→      { id: '1', label: 'Understanding request', status: 'pending' },
   321→      { id: '2', label: 'Reading files', status: 'pending' },
   322→      { id: '3', label: 'Planning artifact', status: 'pending' },
   323→      { id: '4', label: 'Generating content', status: 'pending' },
   324→      { id: '5', label: 'Creating visuals', status: 'pending' },
   325→      { id: '6', label: 'Formatting document', status: 'pending' },

   327→      { id: '8', label: 'Finalizing', status: 'pending' },

   330→
   331→    try {
   332→      // Call actual API endpoint
   333→      const response = await fetch('/api/artifacts/generate', {

   335→        headers: { 'Content-Type': 'application/json' },
   336→        body: JSON.stringify({
   337→          prompt,
   338→          artifactType: detectArtifactType(prompt),
   339→          outputFormat: selectedFormat === 'auto' ? undefined : selectedFormat,
   340→          files: files.map(f => ({ name: f.name, size: f.size, type: f.type })),
   341→          workspaceId: user.id, // Use user's default workspace
   342→          brandConfig: null,
   343→          knowledgeContext: null,
   344→        }),
   345→      })
   346→
   347→      if (!response.ok) {
   348→        const errorData = await response.json()
   349→        throw new Error(errorData.message || 'Generation failed')
   350→      }
   351→
   352→      const data = await response.json()
   353→
   354→      // Simulate progress stages based on actual processing
   355→      for (let i = 0; i < stages.length; i++) {
   356→        await new Promise(resolve => setTimeout(resolve, 600 + Math.random() * 800))
   357→        
   358→        setGenerationStages(prev => 
   359→          prev.map(stage => 
   360→            stage.id === String(i + 1) 
   361→              ? { ...stage, status: 'completed' as const }
   362→              : stage
   363→          )
   364→        )
   365→        
   366→        if (i < stages.length - 1) {
   367→          setGenerationStages(prev =>
   368→            prev.map(stage =>
   369→              stage.id === String(i + 2)
   370→                ? { ...stage, status: 'active' as const }
   371→                : stage
   372→            )
   373→          )
   374→        }
   375→      }
   376→
   377→      // Create artifact preview from response or fallback
   378→      const artifact: ArtifactPreview = data.artifact || {
   379→        id: crypto.randomUUID(),
   380→        title: extractTitle(prompt),
   381→        type: detectArtifactType(prompt),
   382→        format: selectedFormat === 'auto' ? 'DOCX' : selectedFormat,
   383→        status: 'completed' as const,
   384→        createdAt: new Date(),
   385→      }
   386→
   387→      setCurrentArtifact(artifact)
   388→      setShowResultDialog(true)
   389→      
   390→      // Clear draft after successful generation
   391→      localStorage.removeItem('filo_draft_prompt')
   392→      
   393→    } catch (err) {
   394→      console.error('Generation failed:', err)
   395→      const errorMessage = err instanceof Error ? err.message : 'Generation failed'
   396→      setError(errorMessage)
   397→      
   398→      // Mark current stage as error
   399→      setGenerationStages(prev =>
   400→        prev.map(stage =>
   401→          stage.status === 'active'
   402→            ? { ...stage, status: 'error' as const, detail: errorMessage }
   403→            : stage
   404→        )
   405→      )
   406→    } finally {
   407→      setIsGenerating(false)
   408→    }
   409→  }
   410→

   415→  }

   420→    if (lower.includes('lesson') || lower.includes('course')) return 'Lesson Plan'

   422→    if (lower.includes('resume') || lower.includes('cv')) return 'Resume'
   423→    if (lower.includes('presentation') || lower.includes('slide')) return 'Presentation'
   424→    if (lower.includes('spreadsheet') || lower.includes('excel') || lower.includes('budget')) return 'Spreadsheet'
   425→    if (lower.includes('report')) return 'Report'
   426→    return 'Document'
   427→  }
   428→

   431→  }
   432→

   434→    setShowLoginModal(false)
   435→    setShowSignupModal(true)
   436→    setError(null)
   437→  }
   438→
   439→  const switchToLogin = () => {
   440→    setShowSignupModal(false)
   441→    setShowLoginModal(true)
   442→    setError(null)
   443→  }
   444→
   445→  return (

   447→      {/* Header with Auth */}
   448→      <header className="sticky top-0 z-40 border-b bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/60">
   449→        <div className="container mx-auto flex h-16 items-center justify-between px-4">

   478→                  <LogIn className="h-4 w-4" />
   479→                  <span className="hidden sm:inline">Log in</span>
   480→                </Button>
   481→                <Button 
   482→                  size="sm" 
   483→                  onClick={() => setShowSignupModal(true)}
   484→                  className="gap-2 cursor-pointer"
   485→                >
   486→                  <UserPlus className="h-4 w-4" />
   487→                  Sign up
   488→                </Button>
   489→              </div>
   490→            )}
   491→          </div>
   492→        </div>
   493→      </header>
   494→
   495→      {/* Hero Section */}
   496→      <section className="relative overflow-hidden border-b bg-gradient-to-br from-background via-background to-muted/30">
   497→        <div className="absolute inset-0 bg-[linear-gradient(to_right,#8882_1px,transparent_1px),linear-gradient(to_bottom,#8882_1px,transparent_1px)] bg-[size:24px_24px] [mask-image:radial-gradient(ellipse_60%_50%_at_50%_0%,#000_70%,transparent_100%)]" />
   498→        
   499→        <div className="container relative mx-auto px-4 py-12 lg:py-20">
   500→          <div className="mx-auto max-w-4xl text-center">
   501→            {/* Badge */}

   503→              variant="secondary" 
   504→              className="mb-6 px-4 py-1.5 text-sm cursor-default select-none"
   505→            >
   506→              <Sparkles className="mr-2 h-3.5 w-3.5" />
   507→              AI-Powered Artifact Generation
   508→            </Badge>
   509→
   510→            {/* Headline */}
   511→            <h1 className="mb-6 text-4xl font-bold tracking-tight sm:text-5xl lg:text-6xl leading-tight">
   512→              Describe what you need.
   513→              <br />
   514→              <span className="bg-gradient-to-r from-primary to-primary/70 bg-clip-text text-transparent">
   515→                Filo creates it.
   516→              </span>
   517→            </h1>

   519→            {/* Subheadline */}

   522→            </p>
   523→
   524→            {/* Error Message */}

   526→              <div className="mx-auto mb-6 max-w-3xl flex items-center gap-3 rounded-lg border border-red-200 bg-red-50 p-4 dark:border-red-800 dark:bg-red-950">

   529→                <button 
   530→                  onClick={() => setError(null)}
   531→                  className="ml-auto p-1 hover:bg-red-100 dark:hover:bg-red-900 rounded transition-colors cursor-pointer"
   532→                >
   533→                  <X className="h-4 w-4" />
   534→                </button>
   535→              </div>
   536→            )}
   537→
   538→            {/* Main Input */}
   539→            <div className="mx-auto max-w-3xl">
   540→              <div className="rounded-xl border bg-card p-2 shadow-lg">
   541→                <Textarea
   542→                  placeholder="What do you want to create? Be specific about what you need..."
   543→                  value={prompt}
   544→                  onChange={(e) => setPrompt(e.target.value)}
   545→                  className="min-h-[120px] resize-none border-0 bg-transparent text-base focus-visible:ring-0 focus-visible:ring-offset-0 px-3 py-3 cursor-text"

   547→                />

   549→                {/* Files area */}

   552→                    className={`mx-2 mb-2 rounded-lg border-2 border-dashed p-4 transition-all duration-200 ${

   579→                        ))}
   580→                        <label>

   582→                            + Add more files
   583→                          </span>
   584→                          <input
   585→                            type="file"
   586→                            multiple
   587→                            className="hidden"
   588→                            onChange={(e) => handleFiles(e.target.files)}
   589→                            disabled={isGenerating}
   590→                          />
   591→                        </label>
   592→                      </div>
   593→                    ) : (
   594→                      <div className="flex flex-col items-center justify-center py-8 text-sm text-muted-foreground">
