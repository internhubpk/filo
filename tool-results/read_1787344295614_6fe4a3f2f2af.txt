   176→  const { theme, setTheme } = useTheme()

   198→    try { await fetch('/api/auth/admin/logout', { method: 'DELETE' }) }
   199→    catch (e) { console.error(e) }
   200→    window.location.href = '/admin/login'
   201→  }
   202→

   204→  const handleCreate = () => {
   205→    setEditingPlan(null)
   206→    setFormData({ ...emptyForm, features: [] })

   209→    setIsDialogOpen(true)
   210→  }

   213→    setEditingPlan(plan)

   215→      name: plan.name,
   216→      description: plan.description || '',
   217→      priceMonthly: plan.priceMonthly,
   218→      priceYearly: plan.priceYearly,
   219→      features: [...(plan.features || [])],
   220→      limitations: [...(plan.limitations || [])],
   221→      popular: plan.popular || false,
   222→      active: plan.active !== false,
   223→      maxAiGenerations: plan.maxAiGenerations || 0,
   224→      maxStorageMb: plan.maxStorageMb || 0,
   225→      icon: plan.icon || 'Crown',
   226→      order: plan.order || 0,

   235→    setDeleteConfirmOpen(true)
   236→  }

   239→    if (!planToDelete) return
   240→    setPlans(prev => prev.filter(p => p._id !== planToDelete._id))

   243→    setPlanToDelete(null)

   249→      name: `${plan.name} (Copy)`,

   284→        active: formData.active,
   285→        maxAiGenerations: formData.maxAiGenerations,
   286→        maxStorageMb: formData.maxStorageMb,
   287→        icon: formData.icon,
   288→        order: formData.order,

   318→
   319→  // Feature/Limitation helpers

   321→    if (newFeature.trim()) {
   322→      setFormData(prev => ({ ...prev, features: [...prev.features, newFeature.trim()] }))
   323→      setNewFeature('')
   324→    }
   325→  }
   326→
   327→  const removeFeature = (idx: number) => {
   328→    setFormData(prev => ({ ...prev, features: prev.features.filter((_, i) => i !== idx) }))
   329→  }
   330→

   332→    if (newLimitation.trim()) {
   333→      setFormData(prev => ({ ...prev, limitations: [...(prev.limitations || []), newLimitation.trim()] }))
   334→      setNewLimitation('')
   335→    }
   336→  }
   337→
   338→  const removeLimitation = (idx: number) => {
   339→    setFormData(prev => ({ ...prev, limitations: prev.limitations?.filter((_, i) => i !== idx) || [] }))

   358→              <span className="text-base md:text-lg font-semibold sm:hidden">Admin</span>
   359→            </div>
   360→          </div>
   361→
   362→          <div className="flex items-center gap-1 md:gap-2">
   363→            <Badge variant="secondary" className="hidden sm:flex gap-1 cursor-default bg-green-50 dark:bg-green-950 text-green-700 dark:text-green-300 border-green-200 dark:border-green-800 text-xs px-2 py-0.5">
   364→              <Shield className="h-3 w-3" /> Admin
   365→            </Badge>
   366→
   367→            <Button variant="ghost" size="icon" onClick={() => setTheme(theme === 'dark' ? 'light' : 'dark')} className="cursor-pointer h-8 w-8 md:h-9 md:w-9">
   368→              <Sun className="h-4 w-4 rotate-0 scale-100 transition-all dark:-rotate-90 dark:scale-0" />
   369→              <Moon className="absolute h-4 w-4 rotate-90 scale-0 transition-all dark:rotate-0 dark:scale-100" />
   370→            </Button>
   371→
   372→            <Button variant="outline" size="sm" onClick={handleLogout} className="gap-1.5 md:gap-2 cursor-pointer text-red-600 hover:text-red-700 hover:bg-red-50 border-red-200 dark:border-red-800 text-xs px-2 md:px-3 h-8 md:h-9">
   373→              <LogOut className="h-3.5 w-3.5" />
   374→              <span className="hidden sm:inline">Logout</span>
   375→            </Button>
   376→
   377→            <Button variant="ghost" size="sm" asChild className="cursor-pointer gap-1.5 md:gap-2">
   378→              <Link href="/"><ArrowLeft className="h-4 w-4" /><span className="hidden sm:inline">Back</span></Link>
   379→            </Button>
   380→          </div>
   381→        </div>
   382→      </header>
   383→
   384→      {/* Notification */}
   385→      {notification && (
   386→        <div className={`fixed top-20 right-4 z-50 flex items-center gap-3 px-4 py-3 rounded-lg shadow-lg border animate-in slide-in-from-right max-w-sm mx-4 ${
   387→          notification.type === 'success' 
   388→            ? 'bg-green-50 dark:bg-green-950 border-green-200 dark:border-green-800 text-green-800 dark:text-green-200' 
   389→            : 'bg-red-50 dark:bg-red-950 border-red-200 dark:border-red-800 text-red-800 dark:text-red-200'
   390→        }`}>
   391→          {notification.type === 'success' ? <Check className="h-5 w-5 shrink-0" /> : <AlertCircle className="h-5 w-5 shrink-0" />}
   392→          <span className="font-medium text-sm">{notification.message}</span>
   393→        </div>
   394→      )}
   395→
   396→      <main className="container mx-auto px-4 py-6 md:py-8">
   397→        {/* Demo Mode Banner */}
   398→        <Card className="mb-6 border-yellow-200 dark:border-yellow-800 bg-yellow-50 dark:bg-yellow-950">
   399→          <CardContent className="pt-4 pb-4 px-4">
   400→            <div className="flex flex-col sm:flex-row items-start sm:items-center gap-3">
   401→              <Database className="h-5 w-5 text-yellow-600 shrink-0 mt-0.5" />
   402→              <div className="flex-1 min-w-0">
   403→                <div className="flex items-center gap-2 mb-1">
   404→                  <h3 className="font-semibold text-sm text-yellow-800 dark:text-yellow-200">Demo Mode</h3>
   405→                  <Badge variant="outline" className="text-xs border-yellow-300 dark:border-yellow-700">Local Storage</Badge>
   406→                </div>
   407→                <p className="text-xs text-yellow-700 dark:text-yellow-300">
   408→                  Connect Convex for persistent database storage.
   409→                </p>
   410→              </div>
   411→            </div>
   412→          </CardContent>
   413→        </Card>
   414→
   415→        {/* Page Header */}
   416→        <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4 mb-6">
   417→          <div>

   422→          <div className="flex flex-wrap items-center gap-2">

   426→                  <Plus className="h-4 w-4" /> New Plan

   428→              </DialogTrigger>
   429→              
   430→              <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto mx-4 sm:mx-0">
   431→                <DialogHeader>
   432→                  <DialogTitle>{editingPlan ? 'Edit Plan' : 'Create Plan'}</DialogTitle>

   482→                  <div className="flex flex-wrap gap-4">

   486→                    </label>
   487→                    <label className="flex items-center gap-2 cursor-pointer">

   489→                      <span>Popular</span>
   490→                    </label>
   491→                  </div>
